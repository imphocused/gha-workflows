# gha_workflows

Centralized reusable GitHub Actions workflows for DevSecNetOps repositories.

## Purpose

This repo provides shared CI policy/workflow logic so all service repos inherit:

- consistent lint/security checks
- audit evidence artifact upload
- least-privilege CI defaults
- policy enforcement (SHA pinning, workflow standards)

## Primary reusable workflow

- `.github/workflows/compliance-reusable.yml`

Used via:

```yaml
jobs:
  compliance:
    uses: your-org/gha_workflows/.github/workflows/compliance-reusable.yml@<PINNED_SHA>
    with:
      profile: terraform # terraform|ansible|packer|scripts
```

## Security and hardening expectations

- All `uses:` must be pinned to immutable commit SHAs.
- Workflow-level `permissions: {}` is required.
- Job-level permissions must be minimal.
- Concurrency controls are required to prevent duplicate runs.
- Audit evidence must be uploaded for every run (`if: always()`).

## Versioning model

Recommended:
1. Merge reviewed changes to `main`.
2. Tag reviewed release points (`v1`, `v1.1.0`, etc.).
3. Consumer repos pin to commit SHA from release commit.
4. Dependabot in consumer repos updates pinned SHAs weekly.

## Local validation

```bash
# Optional local tools
pip install actionlint
actionlint
```

## Operational notes

- Do not use floating refs (`@main`, `@v4`) in production workflows.
- Do not widen permissions without documented justification.
- Any breaking changes require migration notes in PR description.
EOF

write_file "${BASE_DIR}/.github/workflows/compliance-reusable.yml" <<'EOF'
name: compliance-reusable

on:
  workflow_call:
    inputs:
      profile:
        description: "terraform | ansible | packer | scripts"
        required: true
        type: string
      working_directory:
        description: "Directory where checks run"
        required: false
        default: "."
        type: string
      python_version:
        required: false
        default: "3.11"
        type: string
      terraform_version:
        required: false
        default: "1.8.5"
        type: string
      packer_version:
        required: false
        default: "1.11.2"
        type: string
      node_version:
        required: false
        default: "20"
        type: string
      ruby_version:
        required: false
        default: "3.2"
        type: string
      retention_days:
        required: false
        default: 90
        type: number
      run_precommit:
        required: false
        default: true
        type: boolean
      run_detect_secrets:
        required: false
        default: true
        type: boolean

permissions: {}

jobs:
  compliance:
    runs-on: ubuntu-latest
    timeout-minutes: 30

    concurrency:
      group: compliance-${{ github.repository }}-${{ inputs.profile }}-${{ github.ref }}-${{ inputs.working_directory }}
      cancel-in-progress: ${{ github.event_name == 'pull_request' }}

    permissions:
      contents: read
      actions: write

    steps:
      - name: Checkout
        uses: actions/checkout@REPLACE_WITH_CHECKOUT_SHA
        with:
          fetch-depth: 1
          persist-credentials: false

      - name: Validate profile input
        shell: bash
        run: |
          set -euo pipefail
          case "${{ inputs.profile }}" in
            terraform|ansible|packer|scripts) ;;
            *) echo "Invalid profile: ${{ inputs.profile }}"; exit 1 ;;
          esac

      - name: Create audit evidence directory
        run: mkdir -p "${{ inputs.working_directory }}/audit-evidence"

      - name: Verify required config files
        shell: bash
        run: |
          set -euo pipefail
          cd "${{ inputs.working_directory }}"
          test -f .pre-commit-config.yaml
          if [[ "${{ inputs.run_detect_secrets }}" == "true" ]]; then
            test -f .secrets.baseline
          fi
          case "${{ inputs.profile }}" in
            terraform) test -f .tflint.hcl && test -f .tfsec.yml ;;
            ansible)   test -f .yamllint && test -f .ansible-lint ;;
            packer)    test -f tools/precommit/packer-fmt.sh && test -f tools/precommit/packer-validate.sh ;;
            scripts)   true ;;
          esac

      - name: Setup Python
        uses: actions/setup-python@REPLACE_WITH_SETUP_PYTHON_SHA
        with:
          python-version: ${{ inputs.python_version }}

      - name: Install base tooling
        run: pip install --disable-pip-version-check --no-input pre-commit detect-secrets

      - name: Setup Terraform
        if: inputs.profile == 'terraform'
        uses: hashicorp/setup-terraform@REPLACE_WITH_SETUP_TERRAFORM_SHA
        with:
          terraform_version: ${{ inputs.terraform_version }}

      - name: Setup TFLint
        if: inputs.profile == 'terraform'
        uses: terraform-linters/setup-tflint@REPLACE_WITH_SETUP_TFLINT_SHA

      - name: Install tfsec
        if: inputs.profile == 'terraform'
        run: curl -sSfL https://raw.githubusercontent.com/aquasecurity/tfsec/master/scripts/install_linux.sh | bash

      - name: Install Ansible linting tools
        if: inputs.profile == 'ansible'
        run: pip install --disable-pip-version-check --no-input ansible-core ansible-lint yamllint

      - name: Setup Packer
        if: inputs.profile == 'packer'
        uses: hashicorp/setup-packer@REPLACE_WITH_SETUP_PACKER_SHA
        with:
          version: ${{ inputs.packer_version }}

      - name: Setup Node
        if: inputs.profile == 'scripts'
        uses: actions/setup-node@REPLACE_WITH_SETUP_NODE_SHA
        with:
          node-version: ${{ inputs.node_version }}

      - name: Setup Ruby
        if: inputs.profile == 'scripts'
        uses: ruby/setup-ruby@REPLACE_WITH_SETUP_RUBY_SHA
        with:
          ruby-version: ${{ inputs.ruby_version }}

      - name: Terraform checks
        if: inputs.profile == 'terraform'
        shell: bash
        run: |
          set -euo pipefail
          cd "${{ inputs.working_directory }}"
          terraform fmt -check -recursive |& tee audit-evidence/terraform-fmt.txt
          terraform init -backend=false -lockfile=readonly |& tee audit-evidence/terraform-init.txt
          terraform validate |& tee audit-evidence/terraform-validate.txt
          tflint --init |& tee audit-evidence/tflint-init.txt
          tflint --config .tflint.hcl --recursive --format compact |& tee audit-evidence/tflint.txt
          tfsec --config-file .tfsec.yml --format json . > audit-evidence/tfsec.json
          tfsec --config-file .tfsec.yml --no-color . |& tee audit-evidence/tfsec.txt

      - name: Ansible checks
        if: inputs.profile == 'ansible'
        shell: bash
        run: |
          set -euo pipefail
          cd "${{ inputs.working_directory }}"
          yamllint -f parsable -c .yamllint . |& tee audit-evidence/yamllint.txt
          ansible-lint -c .ansible-lint -p . |& tee audit-evidence/ansible-lint.txt

      - name: Packer checks
        if: inputs.profile == 'packer'
        shell: bash
        run: |
          set -euo pipefail
          cd "${{ inputs.working_directory }}"
          packer fmt -check . |& tee audit-evidence/packer-fmt.txt
          packer init . |& tee audit-evidence/packer-init.txt
          packer validate . |& tee audit-evidence/packer-validate.txt

      - name: Scripts optional yamllint
        if: inputs.profile == 'scripts'
        shell: bash
        run: |
          set -euo pipefail
          cd "${{ inputs.working_directory }}"
          if [[ -f .yamllint ]]; then
            pip install --disable-pip-version-check --no-input yamllint
            yamllint -f parsable -c .yamllint . |& tee audit-evidence/yamllint.txt
          else
            echo ".yamllint not present" | tee audit-evidence/yamllint.txt
          fi

      - name: Pre-commit check
        if: inputs.run_precommit
        shell: bash
        run: |
          set -euo pipefail
          cd "${{ inputs.working_directory }}"
          pre-commit run --all-files --show-diff-on-failure |& tee audit-evidence/pre-commit.txt

      - name: Secrets baseline drift check
        if: inputs.run_detect_secrets
        shell: bash
        run: |
          set -euo pipefail
          cd "${{ inputs.working_directory }}"
          tmpfile="$(mktemp)"
          detect-secrets scan $(git ls-files) > "${tmpfile}"
          diff -u .secrets.baseline "${tmpfile}" |& tee audit-evidence/detect-secrets-drift.txt

      - name: Fail if tools introduced drift
        shell: bash
        run: |
          set -euo pipefail
          git update-index -q --refresh
          git diff --exit-code |& tee "${{ inputs.working_directory }}/audit-evidence/git-drift.txt"

      - name: Upload audit evidence
        if: always()
        uses: actions/upload-artifact@REPLACE_WITH_UPLOAD_ARTIFACT_SHA
        with:
          name: "${{ inputs.profile }}-audit-evidence-${{ github.run_id }}"
          path: "${{ inputs.working_directory }}/audit-evidence/"
          if-no-files-found: warn
          retention-days: ${{ inputs.retention_days }}
EOF

write_file "${BASE_DIR}/.github/workflows/policy-enforce-sha-pinning.yml" <<'EOF'
name: policy-enforce-sha-pinning

on:
  pull_request:
    paths:
      - ".github/workflows/**"
      - ".github/actions/**"
      - "action.yml"
      - "action.yaml"
  push:
    branches: [main]
    paths:
      - ".github/workflows/**"
      - ".github/actions/**"
      - "action.yml"
      - "action.yaml"

permissions: {}

jobs:
  enforce-sha-pinning:
    runs-on: ubuntu-latest
    permissions:
      contents: read

    steps:
      - name: Checkout
        uses: actions/checkout@8ade135a41bc03ea155e62e844d188df1ea18608

      - name: Enforce immutable `uses:` refs
        shell: bash
        run: |
          set -euo pipefail
          python3 - <<'PY'
          import pathlib, re, sys

          files = list(pathlib.Path(".github/workflows").rglob("*.yml"))
          files += list(pathlib.Path(".github/workflows").rglob("*.yaml"))

          uses_re = re.compile(r'^\s*uses\s*:\s*(.+?)\s*$', re.IGNORECASE)
          sha40_re = re.compile(r'^[0-9a-fA-F]{40}$')
          docker_digest_re = re.compile(r'^docker://.+@sha256:[0-9a-fA-F]{64}$')

          violations = []
          for f in sorted(set(files)):
              for i, raw in enumerate(f.read_text(encoding="utf-8").splitlines(), 1):
                  line = raw.split(" #", 1)[0].rstrip()
                  m = uses_re.match(line)
                  if not m:
                      continue
                  val = m.group(1).strip().strip("'").strip('"')

                  # Local actions/reusable workflows are allowed.
                  if val.startswith("./") or val.startswith("../"):
                      continue

                  # Block dynamic refs.
                  if "${{" in val:
                      violations.append((f, i, val, "dynamic ref not allowed"))
                      continue

                  # Docker refs must be digest-pinned.
                  if val.startswith("docker://"):
                      if not docker_digest_re.match(val):
                          violations.append((f, i, val, "docker must use @sha256 digest"))
                      continue

                  # Remote uses must include @<40-char-sha>.
                  if "@" not in val:
                      violations.append((f, i, val, "missing @ref"))
                      continue

                  ref = val.rsplit("@", 1)[1]
                  if not sha40_re.fullmatch(ref):
                      violations.append((f, i, val, "ref is not 40-char commit SHA"))

          if violations:
              print("❌ SHA pinning violations detected:")
              for f, i, v, r in violations:
                  print(f"- {f}:{i}: {v} ({r})")
              sys.exit(1)

          print("✅ All `uses:` entries are immutably pinned.")
          PY
EOF

name: policy-enforce-workflow-standards

on:
  pull_request:
    paths:
      - ".github/workflows/**"
  push:
    branches: [main]
    paths:
      - ".github/workflows/**"

permissions: {}

jobs:
  enforce-standards:
    runs-on: ubuntu-latest
    permissions:
      contents: read

    steps:
      - name: Checkout
        uses: actions/checkout@8ade135a41bc03ea155e62e844d188df1ea18608

      - name: Install PyYAML dependency
        run: pip install --disable-pip-version-check --no-input pyyaml

      - name: Enforce workflow baseline standards
        shell: bash
        run: |
          set -euo pipefail
          python3 - <<'PY'
          import pathlib
          import sys
          import yaml

          workflow_files = sorted(pathlib.Path(".github/workflows").glob("*.yml"))
          workflow_files += sorted(pathlib.Path(".github/workflows").glob("*.yaml"))

          violations = []

          for wf in workflow_files:
              data = yaml.safe_load(wf.read_text(encoding="utf-8")) or {}

              # 1) Require top-level permissions.
              if "permissions" not in data:
                violations.append((wf, "missing top-level permissions (expect permissions: {})"))

              # 2) Require at least one job.
              jobs = data.get("jobs", {})
              if not jobs:
                violations.append((wf, "missing jobs"))
                continue

              # 3) Require concurrency for each job.
              for job_name, job in jobs.items():
                if "concurrency" not in job:
                  violations.append((wf, f"job '{job_name}' missing concurrency"))

              # 4) Discourage pull_request_target by policy.
              on = data.get("on")
              on_text = str(on)
              if "pull_request_target" in on_text:
                violations.append((wf, "uses pull_request_target; requires explicit security review"))

          if violations:
            print("❌ Workflow standards violations:")
            for wf, msg in violations:
              print(f"- {wf}: {msg}")
            sys.exit(1)

          print("✅ Workflow standards policy passed.")
          PY
EOF
